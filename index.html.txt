<script>
const ACTIONS = ["Спорт","Питание","Чтение перед сном","Английский","Копилка"];
const TAGS = ["1 час","план","1 час","15 мин","сумма"];
const KEY = "100days_v1";

let state = load();

function load(){
  try{
    const raw = localStorage.getItem(KEY);
    if(raw){
      const s = JSON.parse(raw);
      // миграция: если действий было 5 и они старые — оставим совместимость
      return s;
    }
  }catch(e){}
  return { startDate: todayStr(), currentDay: 1, days: {} };
}
function save(){ localStorage.setItem(KEY, JSON.stringify(state)); }
function todayStr(){ return new Date().toISOString().slice(0,10); }

function getDay(n){
  if(!state.days[n]) state.days[n] = { marks: ACTIONS.map(()=>0), note:"", done:false, piggy:0 };
  if(typeof state.days[n].piggy !== "number") state.days[n].piggy = 0;
  return state.days[n];
}

function render(){
  const day = state.currentDay;
  const d = getDay(day);
  document.getElementById("day").textContent = day;
  document.getElementById("left").textContent = Math.max(0, 100 - day);
  document.getElementById("progress").style.width = (day/100*100) + "%";
  document.getElementById("note").value = d.note || "";

  /* Задания */
  const box = document.getElementById("tasks");
  box.innerHTML = "";
  ACTIONS.forEach((name, i)=>{
    const row = document.createElement("div");
    row.className = "task";

    const check = document.createElement("button");
    check.className = "check" + (d.marks[i] > 0 ? " done" : "");
    check.textContent = d.marks[i] > 0 ? "✓" : "";
    check.onclick = ()=>{ d.marks[i] = d.marks[i] > 0 ? 0 : 5; save(); render(); };

    const span = document.createElement("span");
    span.className = "name"; span.textContent = name;

    const tag = document.createElement("span");
    tag.className = "tag"; tag.textContent = TAGS[i];

    const stars = document.createElement("div");
    stars.className = "stars";
    for(let s=1; s<=5; s++){
      const st = document.createElement("span");
      st.className = "star" + (d.marks[i] >= s ? " on" : "");
      st.textContent = "★";
      st.onclick = (e)=>{ e.stopPropagation(); d.marks[i] = s; save(); render(); };
      stars.appendChild(st);
    }

    row.append(check, span, tag, stars);
    box.appendChild(row);
  });

  /* Эффективность дня */
  const total = d.marks.reduce((a,b)=>a+b,0);
  const percent = Math.round(total / 25 * 100);
  document.getElementById("dayScore").textContent = percent + " / 100";

  const dayStars = document.getElementById("dayStars");
  dayStars.innerHTML = "";
  const avgStars = Math.round(percent / 20);
  for(let s=1; s<=5; s++){
    const st = document.createElement("span");
    st.className = "star" + (avgStars >= s ? " on" : "");
    st.style.fontSize = "28px";
    st.textContent = "★";
    st.onclick = ()=>{
      const val = s * 20;
      const base = Math.floor(val/5);
      d.marks = d.marks.map(()=>base);
      let rest = val - base*5;
      for(let i=0; i<d.marks.length && rest>0; i++){ d.marks[i]++; rest--; }
      save(); render();
    };
    dayStars.appendChild(st);
  }

  /* Копилка */
  renderPiggy();

  /* Статистика */
  document.getElementById("avg7").textContent = avgPercent(7) + "%";
  document.getElementById("avg30").textContent = avgPercent(30) + "%";
  document.getElementById("avgAll").textContent = avgPercent(1000) + "%";
  document.getElementById("streak").textContent = calcStreak() + " 🔥";
  document.getElementById("aiText").textContent = buildAiText();
}

/* ---------- КОПИЛКА ---------- */
function renderPiggy(){
  let total = 0;
  const rows = [];
  for(let i=1; i<=100; i++){
    const d = state.days[i];
    if(d && d.piggy > 0){
      total += d.piggy;
      rows.push({day:i, sum:d.piggy});
    }
  }
  document.getElementById("piggyTotal").textContent = fmt(total) + " ₽";

  const list = document.getElementById("piggyList");
  list.innerHTML = "";
  if(rows.length === 0){
    list.innerHTML = '<div style="border:0;opacity:.7">Пока пусто — начни с первой суммы.</div>';
  }else{
    // последние 10 записей сверху
    rows.slice(-10).reverse().forEach(r=>{
      const el = document.createElement("div");
      el.innerHTML = `<span>День ${r.day}</span><b>${fmt(r.sum)} ₽</b>`;
      list.appendChild(el);
    });
  }
  const inp = document.getElementById("piggyInput");
  const cur = getDay(state.currentDay).piggy;
  inp.value = cur > 0 ? cur : "";
}

function addPiggy(){
  const inp = document.getElementById("piggyInput");
  const val = Math.max(0, Math.floor(Number(inp.value) || 0));
  const d = getDay(state.currentDay);
  d.piggy = val;
  save();
  renderPiggy();
}

function fmt(n){ return n.toLocaleString("ru-RU"); }

/* ---------- СТАТИСТИКА ---------- */
function avgPercent(n){
  let sum=0, cnt=0;
  for(let i=1; i<=Math.min(n, state.currentDay); i++){
    const d = state.days[i];
    if(d && d.marks){
      sum += d.marks.reduce((a,b)=>a+b,0)/25*100;
      cnt++;
    }
  }
  return cnt ? Math.round(sum/cnt) : 0;
}

function calcStreak(){
  let s=0;
  for(let i=state.currentDay; i>=1; i--){
    const d = state.days[i];
    if(d && d.marks && d.marks.some(x=>x>0)) s++;
    else break;
  }
  return s;
}

function buildAiText(){
  const a7 = avgPercent(7), a30 = avgPercent(30);
  if(state.currentDay < 3) return "Пока мало данных. Отмечай выполнение и ставь звёзды — через несколько дней появится анализ.";
  const weakest = weakestAction();
  return `Средняя эффективность за 7 дней — ${a7}%, за 30 дней — ${a30}%. `
       + (weakest ? `Чаще всего проседает «${weakest}». Попробуй сделать это действие минимальным: 5–10 минут или одно конкретное действие в день.` : "");
}

function weakestAction(){
  const sums = ACTIONS.map(()=>0), cnts = ACTIONS.map(()=>0);
  for(let i=1; i<=state.currentDay; i++){
    const d = state.days[i];
    if(!d || !d.marks) continue;
    d.marks.forEach((m,idx)=>{ sums[idx]+=m; cnts[idx]++; });
  }
  let min=Infinity, idx=-1;
  sums.forEach((s,i)=>{ if(cnts[i]){ const a=s/cnts[i]; if(a<min){min=a; idx=i;} } });
  return idx>=0 ? ACTIONS[idx] : null;
}

/* ---------- СОХРАНЕНИЕ / НАВИГАЦИЯ ---------- */
function finishDay(){
  const d = getDay(state.currentDay);
  d.note = document.getElementById("note").value.trim();
  d.done = true;
  d.date = todayStr();
  save();
  const percent = Math.round(d.marks.reduce((a,b)=>a+b,0)/25*100);
  alert("День " + state.currentDay + " сохранён: " + percent + "%");
}

function nextDay(){
  finishDay();
  if(state.currentDay < 100){
    state.currentDay++;
    save();
    render();
    window.scrollTo({top:0, behavior:"smooth"});
  } else {
    alert("Поздравляю! 100 дней завершены 🎉");
  }
}

function resetDay(){
  if(confirm("Сбросить отметки сегодняшнего дня?")){
    state.days[state.currentDay] = { marks: ACTIONS.map(()=>0), note:"", done:false, piggy:0 };
    save(); render();
  }
}

/* ---------- ЭКСПОРТ / ИМПОРТ ---------- */
function exportJSON(){
  const data = {
    app: "100days",
    version: 2,
    exportedAt: new Date().toISOString(),
    state: state
  };
  const blob = new Blob([JSON.stringify(data, null, 2)], {type:"application/json"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  const stamp = new Date().toISOString().slice(0,10);
  a.href = url;
  a.download = "100days-backup-" + stamp + ".json";
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

function importJSON(ev){
  const file = ev.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = (e)=>{
    try{
      const parsed = JSON.parse(e.target.result);
      const incoming = parsed.state ? parsed.state : parsed;
      if(!incoming.days || typeof incoming.currentDay !== "number"){
        throw new Error("Неверный формат файла");
      }
      if(!confirm("Импорт заменит текущий прогресс. Продолжить?")) return;
      state = {
        startDate: incoming.startDate || todayStr(),
        currentDay: Math.min(100, Math.max(1, incoming.currentDay)),
        days: incoming.days || {}
      };
      save(); render();
      alert("Импорт выполнен успешно ✅");
    }catch(err){
      alert("Не удалось прочитать файл: " + err.message);
    }finally{
      ev.target.value = "";
    }
  };
  reader.readAsText(file);
}

function wipeAll(){
  if(confirm("Удалить весь прогресс безвозвратно?")){
    localStorage.removeItem(KEY);
    state = { startDate: todayStr(), currentDay: 1, days: {} };
    save(); render();
    window.scrollTo({top:0, behavior:"smooth"});
  }
}
